1//! Sqlite adapter of `IndexStore` — single file, zero server,
2//! rebuildable. Lives at `.kuda/index.sqlite` (gitignored).
3
4use kuda_core::{DepEdge, EdgeKind, Symbol, SymbolKind};
5use kuda_store::IndexStore;
6use rusqlite::{params, Connection};
7use std::path::Path;
8use std::sync::Mutex;
9
10/// `Connection` is `!Sync` — the mutex keeps the store behind the
11/// port's `Send + Sync` bound. Single-writer anyway.
12pub struct SqliteStore {
13    conn: Mutex<Connection>,
14}
15
16const SCHEMA: &str = "
17CREATE TABLE IF NOT EXISTS symbols(
18    name TEXT NOT NULL,
19    kind TEXT NOT NULL,
20    file TEXT NOT NULL,
21    line INTEGER NOT NULL,
22    module TEXT NOT NULL
23);
24CREATE INDEX IF NOT EXISTS idx_symbols_name ON symbols(name);
25CREATE INDEX IF NOT EXISTS idx_symbols_module ON symbols(module);
26CREATE TABLE IF NOT EXISTS edges(
27    from_mod TEXT NOT NULL,
28    to_mod TEXT NOT NULL,
29    kind TEXT NOT NULL
30);
31CREATE INDEX IF NOT EXISTS idx_edges_to ON edges(to_mod);
32CREATE TABLE IF NOT EXISTS meta(k TEXT PRIMARY KEY, v TEXT NOT NULL);
33";

35fn sk_str(k: SymbolKind) -> &'static str {
36    match k {
37        SymbolKind::Function => "function",
38        SymbolKind::Type => "type",
39        SymbolKind::Trait => "trait",
40        SymbolKind::Const => "const",
41        SymbolKind::Module => "module",
42        SymbolKind::Macro => "macro",
43        SymbolKind::Other => "other",
44    }
45}
46
47fn sk_from(s: &str) -> SymbolKind {
48    match s {
49        "function" => SymbolKind::Function,
50        "type" => SymbolKind::Type,
51        "trait" => SymbolKind::Trait,
52        "const" => SymbolKind::Const,
53        "module" => SymbolKind::Module,
54        "macro" => SymbolKind::Macro,
55        _ => SymbolKind::Other,
56    }
57}
58
59fn ek_str(k: EdgeKind) -> &'static str {
60    match k {
61        EdgeKind::Imports => "imports",
62        EdgeKind::Calls => "calls",
63        EdgeKind::PackageDep => "package-dep",
64        EdgeKind::WorkspaceMember => "workspace-member",
65    }
66}
67
68fn ek_from(s: &str) -> EdgeKind {
69    match s {
70        "imports" => EdgeKind::Imports,
71        "calls" => EdgeKind::Calls,
72        "workspace-member" => EdgeKind::WorkspaceMember,
73        _ => EdgeKind::PackageDep,
74    }
75}

77impl SqliteStore {
78    /// Open (creating) the store at `path`; parents are made.
79    pub fn open(path: &Path) -> Result<Self, String> {
80        if let Some(p) = path.parent() {
81            std::fs::create_dir_all(p).map_err(|e| e.to_string())?;
82        }
83        let conn = Connection::open(path).map_err(|e| e.to_string())?;
84        conn.execute_batch(SCHEMA).map_err(|e| e.to_string())?;
85        Ok(Self {
86            conn: Mutex::new(conn),
87        })
88    }
89
90    fn lock(&self) -> Result<std::sync::MutexGuard<'_, Connection>, String> {
91        self.conn.lock().map_err(|e| e.to_string())
92    }
93}

95impl IndexStore for SqliteStore {
96    fn replace_module(
97        &mut self,
98        module: &str,
99        syms: &[Symbol],
100        edges: &[DepEdge],
101    ) -> Result<(), String> {
102        let mut conn = self.lock()?;
103        let tx = conn.transaction().map_err(|e| e.to_string())?;
104        tx.execute("DELETE FROM symbols WHERE module = ?1", params![module])
105            .and_then(|_| tx.execute("DELETE FROM edges WHERE from_mod = ?1", params![module]))
106            .map_err(|e| e.to_string())?;
107        for s in syms {
108            tx.execute(
109                "INSERT INTO symbols(name,kind,file,line,module) VALUES(?1,?2,?3,?4,?5)",
110                params![
111                    s.name,
112                    sk_str(s.kind),
113                    s.file.to_string_lossy(),
114                    s.line,
115                    s.module
116                ],
117            )
118            .map_err(|e| e.to_string())?;
119        }
120        for e in edges {
121            tx.execute(
122                "INSERT INTO edges(from_mod,to_mod,kind) VALUES(?1,?2,?3)",
123                params![e.from, e.to, ek_str(e.kind)],
124            )
125            .map_err(|e| e.to_string())?;
126        }
127        tx.commit().map_err(|e| e.to_string())
128    }
129
130    fn find(&self, name: &str) -> Result<Vec<Symbol>, String> {
131        let conn = self.lock()?;
132        let mut st = conn
133            .prepare("SELECT name,kind,file,line,module FROM symbols WHERE name = ?1")
134            .map_err(|e| e.to_string())?;
135        let rows = st
136            .query_map(params![name], |r| {
137                Ok(Symbol {
138                    name: r.get(0)?,
139                    kind: sk_from(&r.get::<_, String>(1)?),
140                    file: r.get::<_, String>(2)?.into(),
141                    line: r.get(3)?,
142                    module: r.get(4)?,
143                })
144            })
145            .map_err(|e| e.to_string())?;
146        rows.collect::<Result<Vec<_>, _>>().map_err(|e| e.to_string())
147    }
148
149    fn dependents(&self, name: &str) -> Result<Vec<DepEdge>, String> {
150        let conn = self.lock()?;
151        let mut st = conn
152            .prepare("SELECT from_mod,to_mod,kind FROM edges WHERE to_mod = ?1")
153            .map_err(|e| e.to_string())?;
154        let rows = st
155            .query_map(params![name], |r| {
156                Ok(DepEdge {
157                    from: r.get(0)?,
158                    to: r.get(1)?,
159                    kind: ek_from(&r.get::<_, String>(2)?),
160                })
161            })
162            .map_err(|e| e.to_string())?;
163        rows.collect::<Result<Vec<_>, _>>().map_err(|e| e.to_string())
164    }
165
166    fn modules(&self) -> Result<Vec<String>, String> {
167        let conn = self.lock()?;
168        let mut st = conn
169            .prepare("SELECT DISTINCT module FROM symbols ORDER BY module")
170            .map_err(|e| e.to_string())?;
171        let rows = st.query_map([], |r| r.get(0)).map_err(|e| e.to_string())?;
172        rows.collect::<Result<Vec<_>, _>>().map_err(|e| e.to_string())
173    }
174
175    fn set_meta(&mut self, key: &str, value: &str) -> Result<(), String> {
176        self.lock()?
177            .execute(
178                "INSERT INTO meta(k,v) VALUES(?1,?2) ON CONFLICT(k) DO UPDATE SET v=?2",
179                params![key, value],
180            )
181            .map(|_| ())
182            .map_err(|e| e.to_string())
183    }
184
185    fn meta(&self, key: &str) -> Result<Option<String>, String> {
186        self.lock()?
187            .query_row("SELECT v FROM meta WHERE k = ?1", params![key], |r| r.get(0))
188            .map(Some)
189            .or_else(|e| match e {
190                rusqlite::Error::QueryReturnedNoRows => Ok(None),
191                e => Err(e.to_string()),
192            })
193    }

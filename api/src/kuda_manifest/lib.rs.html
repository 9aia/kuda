1//! Manifest adapter — package manifests -> module graph. No parser,
2//! no language knowledge: reads manifest basenames declared by
3//! `LangDef` entries (Cargo.toml, package.json, go.mod, …).
4
5use kuda_core::{DepEdge, EdgeKind, LangDef, Symbol, SymbolKind};
6use kuda_extract::{Extraction, Extractor};
7use std::collections::HashSet;
8use std::path::Path;
9
10pub struct ManifestExtractor;
11
12impl Extractor for ManifestExtractor {
13    fn kind(&self) -> &'static str {
14        "manifest"
15    }
16
17    fn extract(&self, root: &Path, langs: &[LangDef]) -> Result<Extraction, String> {
18        let names: HashSet<&str> = langs
19            .iter()
20            .flat_map(|l| l.manifests.iter().map(String::as_str))
21            .collect();
22        let mut out = Extraction::default();
23        let mut stack = vec![root.to_path_buf()];
24        while let Some(dir) = stack.pop() {
25            let Ok(entries) = std::fs::read_dir(&dir) else {
26                continue;
27            };
28            for e in entries.flatten() {
29                let path = e.path();
30                let name = e.file_name();
31                let name = name.to_string_lossy();
32                if path.is_dir() {
33                    // skip noise dirs + mounted git worktrees (.git
34                    // as a *file* marks a worktree, not a repo dir)
35                    if !matches!(name.as_ref(), ".git" | "target" | "node_modules" | ".kuda")
36                        && !path.join(".git").is_file()
37                    {
38                        stack.push(path);
39                    }
40                    continue;
41                }
42                if !names.contains(name.as_ref()) {
43                    continue;
44                }
45                let rel = path.strip_prefix(root).unwrap_or(&path).to_path_buf();
46                let text = std::fs::read_to_string(&path).unwrap_or_default();
47                let (pkg, deps) = parse_manifest(name.as_ref(), &text);
48                let module = pkg.unwrap_or_else(|| {
49                    path.parent()
50                        .and_then(|p| p.file_name())
51                        .map(|f| f.to_string_lossy().into_owned())
52                        .unwrap_or_else(|| name.clone().into_owned())
53                });
54                out.symbols.push(Symbol {
55                    name: module.clone(),
56                    kind: SymbolKind::Module,
57                    file: rel.clone(),
58                    line: 1,
59                    module: module.clone(),
60                });
61                for d in deps {
62                    out.edges.push(DepEdge {
63                        from: module.clone(),
64                        to: d,
65                        kind: EdgeKind::PackageDep,
66                    });
67                }
68                out.files.push(rel);
69            }
70        }
71        Ok(out)
72    }
73}

75/// Manifest basename -> (package name, declared deps). Format
76/// detection by filename — adapters are allowed to know concrete tech.
77fn parse_manifest(basename: &str, text: &str) -> (Option<String>, Vec<String>) {
78    match basename {
79        "Cargo.toml" => cargo(text),
80        "package.json" => package_json(text),
81        "go.mod" => go_mod(text),
82        _ => (None, Vec::new()),
83    }
84}
85
86fn cargo(text: &str) -> (Option<String>, Vec<String>) {
87    let Ok(t) = text.parse::<toml::Table>() else {
88        return (None, Vec::new());
89    };
90    let name = t
91        .get("package")
92        .and_then(|p| p.get("name"))
93        .and_then(|n| n.as_str())
94        .map(str::to_owned);
95    let mut deps = Vec::new();
96    for key in ["dependencies", "dev-dependencies", "build-dependencies"] {
97        if let Some(tbl) = t.get(key).and_then(|d| d.as_table()) {
98            deps.extend(tbl.keys().cloned());
99        }
100    }
101    (name, deps)
102}
103
104fn package_json(text: &str) -> (Option<String>, Vec<String>) {
105    let Ok(v) = serde_json::from_str::<serde_json::Value>(text) else {
106        return (None, Vec::new());
107    };
108    let name = v.get("name").and_then(|n| n.as_str()).map(str::to_owned);
109    let mut deps = Vec::new();
110    for key in ["dependencies", "devDependencies"] {
111        if let Some(tbl) = v.get(key).and_then(|d| d.as_object()) {
112            deps.extend(tbl.keys().cloned());
113        }
114    }
115    (name, deps)
116}
117
118fn go_mod(text: &str) -> (Option<String>, Vec<String>) {
119    let name = text
120        .lines()
121        .find_map(|l| l.strip_prefix("module ").map(str::trim).map(str::to_owned));
122    let deps = text
123        .lines()
124        .filter(|l| l.starts_with('\t') && l.contains(" v"))
125        .filter_map(|l| l.split_whitespace().next().map(str::to_owned))
126        .collect();
127    (name, deps)
128}

1//! Treesitter adapter — grammar+query data -> symbols + import edges.
2//! Per-language knowledge lives in `data/langs/*` (`grammar` id,
3//! `extensions`, `symbol_query`); this crate only runs a query
4//! over a grammar it has compiled in. New language = data file +
5//! a `language()` arm — the vocabulary stays fixed.
6
7use kuda_core::{DepEdge, EdgeKind, LangDef, Symbol, SymbolKind};
8use kuda_extract::{Extraction, Extractor};
9use std::path::{Path, PathBuf};
10use tree_sitter::{Language, Parser, Query, QueryCursor, StreamingIterator};
11
12pub struct TreesitterExtractor;
13
14/// Grammar id -> compiled language. Impls know concrete grammars;
15/// ids come from `data/langs/*`.
16fn language(id: &str) -> Option<Language> {
17    match id {
18        "rust" => Some(tree_sitter_rust::LANGUAGE.into()),
19        _ => None,
20    }
21}
22
23/// Default symbol query when a LangDef ships none — captures are
24/// `@name.<kind>` where <kind> maps to `SymbolKind`.
25const DEFAULT_RUST_QUERY: &str = r#"
26(function_item name: (identifier) @name.function)
27(struct_item name: (type_identifier) @name.type)
28(enum_item name: (type_identifier) @name.type)
29(union_item name: (type_identifier) @name.type)
30(trait_item name: (type_identifier) @name.trait)
31(const_item name: (identifier) @name.const)
32(static_item name: (identifier) @name.const)
33(mod_item name: (identifier) @name.module)
34(macro_definition name: (identifier) @name.macro)
35(type_item name: (type_identifier) @name.type)
36"#;
37
38const DEP_QUERY: &str = r#"
39(use_declaration argument: (identifier) @use.root)
40(use_declaration argument: (scoped_identifier path: (identifier) @use.root))
41(use_declaration argument: (scoped_use_list path: (identifier) @use.root))
42"#;
43
44fn kind_of(cap: &str) -> SymbolKind {
45    match cap.rsplit('.').next() {
46        Some("function") => SymbolKind::Function,
47        Some("type") => SymbolKind::Type,
48        Some("trait") => SymbolKind::Trait,
49        Some("const") => SymbolKind::Const,
50        Some("module") => SymbolKind::Module,
51        Some("macro") => SymbolKind::Macro,
52        _ => SymbolKind::Other,
53    }
54}

56/// Nearest ancestor dir (under root) holding a declared manifest —
57/// that dir's name is the module. Fallback: "_root".
58fn module_of(root: &Path, file: &Path, manifests: &[String]) -> String {
59    let mut dir = file.parent();
60    while let Some(d) = dir {
61        if manifests.iter().any(|m| d.join(m).exists()) {
62            return d
63                .file_name()
64                .map(|f| f.to_string_lossy().into_owned())
65                .unwrap_or_else(|| "_root".into());
66        }
67        dir = d.parent();
68        if !d.starts_with(root) {
69            break;
70        }
71    }
72    "_root".into()
73}

75fn walk_files(root: &Path, exts: &[String]) -> Vec<PathBuf> {
76    let mut out = Vec::new();
77    let mut stack = vec![root.to_path_buf()];
78    while let Some(d) = stack.pop() {
79        let Ok(rd) = std::fs::read_dir(&d) else {
80            continue;
81        };
82        for e in rd.flatten() {
83            let p = e.path();
84            if p.is_dir() {
85                if !matches!(
86                    e.file_name().to_str(),
87                    Some(".git" | "target" | "node_modules" | ".kuda")
88                ) && !p.join(".git").is_file()
89                {
90                    stack.push(p);
91                }
92            } else if p
93                .extension()
94                .and_then(|x| x.to_str())
95                .map(|x| exts.iter().any(|e| e.trim_start_matches('.') == x))
96                .unwrap_or(false)
97            {
98                out.push(p);
99            }
100        }
101    }
102    out
103}

105impl Extractor for TreesitterExtractor {
106    fn kind(&self) -> &'static str {
107        "treesitter"
108    }
109
110    fn extract(&self, root: &Path, langs: &[LangDef]) -> Result<Extraction, String> {
111        let mut out = Extraction::default();
112        for lang in langs {
113            let Some(grammar) = lang.grammar.as_deref().and_then(language) else {
114                continue; // no compiled grammar for this lang yet
115            };
116            let src = lang.symbol_query.as_deref().unwrap_or(DEFAULT_RUST_QUERY);
117            let sym_q = Query::new(&grammar, src).map_err(|e| format!("query {e}"))?;
118            let dep_q = Query::new(&grammar, DEP_QUERY).map_err(|e| format!("query {e}"))?;
119            let sym_caps = sym_q.capture_names().to_vec();
120            let dep_caps = dep_q.capture_names().to_vec();
121            let mut parser = Parser::new();
122            parser
123                .set_language(&grammar)
124                .map_err(|e| format!("lang {e}"))?;
125
126            for file in walk_files(root, &lang.extensions) {
127                let Ok(text) = std::fs::read_to_string(&file) else {
128                    continue;
129                };
130                let Some(tree) = parser.parse(&text, None) else {
131                    continue;
132                };
133                let rel = file.strip_prefix(root).unwrap_or(&file).to_path_buf();
134                let module = module_of(root, &file, &lang.manifests);
135                let mut cursor = QueryCursor::new();
136                let mut sym_m = cursor.matches(&sym_q, tree.root_node(), text.as_bytes());
137                while let Some(m) = sym_m.next() {
138                    for cap in m.captures {
139                        let name = &text[cap.node.byte_range()];
140                        out.symbols.push(Symbol {
141                            name: name.to_owned(),
142                            kind: kind_of(sym_caps[cap.index as usize]),
143                            file: rel.clone(),
144                            line: cap.node.start_position().row as u32 + 1,
145                            module: module.clone(),
146                        });
147                    }
148                }
149                let mut dep_m = cursor.matches(&dep_q, tree.root_node(), text.as_bytes());
150                while let Some(m) = dep_m.next() {
151                    for cap in m.captures {
152                        let dep = &text[cap.node.byte_range()];
153                        if matches!(dep, "self" | "crate" | "std" | "core" | "alloc" | "super")
154                            || dep_caps[cap.index as usize].is_empty()
155                        {
156                            continue;
157                        }
158                        out.edges.push(DepEdge {
159                            from: module.clone(),
160                            to: dep.to_owned(),
161                            kind: EdgeKind::Imports,
162                        });
163                    }
164                }
165                out.files.push(rel);
166            }
167        }
168        Ok(out)
169    }
170}

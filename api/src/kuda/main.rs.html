15/// stdout that survives `| head` — BrokenPipe exits 0 like a
16/// well-behaved unix filter instead of panicking.
17fn emit(args: std::fmt::Arguments) {
18    use std::io::Write;
19    let mut o = std::io::stdout().lock();
20    if let Err(e) = o.write_fmt(args).and_then(|_| o.flush()) {
21        if e.kind() == std::io::ErrorKind::BrokenPipe {
22            std::process::exit(0);
23        }
24    }
25}
26macro_rules! out {
27    ($($arg:tt)*) => { emit(format_args!($($arg)*)) };
28}
29macro_rules! outln {
30    ($($arg:tt)*) => { emit(format_args!("{}\n", format_args!($($arg)*))) };
31}

110fn main() {
111    let cli = Cli::parse();
112    let code = match &cli.cmd {
113        Cmd::Init => init(&cli),
114        Cmd::Sync { full } => sync(&cli, *full),
115        Cmd::Status => status(&cli),
116        Cmd::Map => map(&cli),
117        Cmd::Card { module } => card(&cli, module),
118        Cmd::Find { symbol } => find(&cli, symbol),
119        Cmd::Dependents { name } => dependents(&cli, name),
120        Cmd::Refs { name, docs } => refs(&cli, name, *docs),
121        Cmd::Doctor => doctor(&cli),
122        // `watch` runs an unbounded loop and only stops on a signal —
123        // signals discard coverage data, so it's excluded below and
124        // exercised by the e2e test in normal builds.
125        #[cfg(not(coverage))]
126        Cmd::Watch => watch(&cli),
127        #[cfg(coverage)]
128        Cmd::Watch => 0,
129        Cmd::Mcp => mcp(&cli),
130        Cmd::Acp => acp(&cli),
131        Cmd::GenCliDocs => {
132            outln!("{}", clap_markdown::help_markdown::<Cli>());
133            0
134        }
135    };
136    std::process::exit(code);
137}

175/// git blob hash — provenance key. In-process sha1
176/// ("blob <len>\0" + bytes): identical values to
177/// `git hash-object`, no subprocess per file.
178fn hash_files(root: &Path, files: &[PathBuf]) -> Vec<String> {
179    files.iter().map(|f| blob_hash(&root.join(f))).collect()
180}
181
182fn blob_hash(path: &Path) -> String {
183    use sha1::Digest;
184    let Ok(bytes) = std::fs::read(path) else {
185        return String::new();
186    };
187    let mut h = sha1::Sha1::new();
188    h.update(format!("blob {}\0", bytes.len()).as_bytes());
189    h.update(&bytes);
190    h.finalize().iter().map(|b| format!("{b:02x}")).collect()
191}

200fn init(cli: &Cli) -> i32 {
201    let dir = root(cli);
202    let toml_path = dir.join("kuda.toml");
203    if toml_path.exists() {
204        eprintln!("kuda.toml already exists — not overwriting");
205        return 1;
206    }
207    if let Err(e) = std::fs::write(&toml_path, EXAMPLE_TOML) {
208        eprintln!("write {}: {e}", toml_path.display());
209        return 1;
210    }
211    for d in [".kuda", ".kuda/modules"] {
212        if let Err(e) = std::fs::create_dir_all(dir.join(d)) {
213            eprintln!("mkdir {d}: {e}");
214            return 1;
215        }
216    }
217    outln!("wrote {} + .kuda/ — run `kuda sync`", toml_path.display());
218    0
219}

221/// Workspace roots from `.kuda/workspace.yaml` (`roots: [...]`),
222/// resolved relative to the file's dir.
223fn workspace_roots(dir: &Path) -> Vec<PathBuf> {
224    let p = dir.join(".kuda/workspace.yaml");
225    let Ok(text) = std::fs::read_to_string(&p) else {
226        return vec![];
227    };
228    let Ok(v) = serde_yaml::from_str::<serde_yaml::Value>(&text) else {
229        return vec![];
230    };
231    v["roots"]
232        .as_sequence()
233        .map(|s| {
234            s.iter()
235                .filter_map(|r| r.as_str())
236                .map(|r| {
237                    let p = dir.join(r);
238                    std::fs::canonicalize(&p).unwrap_or(p)
239                })
240                .collect()
241        })
242        .unwrap_or_default()
243}

245/// The pipeline: extract -> per-module summarize -> cards + store.
246fn sync(cli: &Cli, full: bool) -> i32 {
247    let dir = root(cli);
248    let cfg = load_config(cli);
249    let roots = if cli.workspace {
250        workspace_roots(&dir)
251    } else {
252        vec![dir.clone()]
253    };
254    let roots = if roots.is_empty() {
255        vec![dir.clone()]
256    } else {
257        roots
258    };
259    // cross-root edge resolution: a dep naming a sibling repo's
260    // module must resolve to its canonical spelling (dash-named
261    // package, underscore import — module_key folds both), not fall
262    // through as external. Names come from each root's manifest
263    // pass + domain descriptors — cheap enough to pre-pass.
264    let union: std::collections::HashSet<String> = if roots.len() > 1 {
265        let ls = langs();
266        roots
267            .iter()
268            .flat_map(|r| {
269                kuda_manifest::ManifestExtractor {
270                    excludes: cfg.extract.exclude.clone(),
271                    resolve_git_deps: cfg.workspace.resolve_git_deps,
272                }
273                .extract(r, &ls)
274                .map(|e| e.symbols.into_iter().map(|s| s.module).collect::<Vec<_>>())
275                .unwrap_or_default()
276            })
277            .chain(domains().iter().map(|d| d.module.clone()))
278            .collect()
279    } else {
280        Default::default()
281    };
282    let mut code = 0;
283    for r in &roots {
284        outln!("sync {}", r.display());
285        code |= sync_one(r, &cfg, cli, full, &union);
286    }
287    code
288}

290/// Walk inputs derived from the loaded langs: manifest basenames
291/// (ordered list + lookup set) and source extensions (dotless).
292fn walk_sets(
293    langs: &[LangDef],
294) -> (
295    Vec<String>,
296    std::collections::HashSet<String>,
297    std::collections::HashSet<String>,
298) {
299    let manifest_list: Vec<String> = langs
300        .iter()
301        .flat_map(|l| l.manifests.iter().cloned())
302        .collect();
303    let manifest_names = manifest_list.iter().cloned().collect();
304    let src_exts = langs
305        .iter()
306        .flat_map(|l| {
307            l.extensions
308                .iter()
309                .map(|e| e.trim_start_matches('.').to_owned())
310        })
311        .collect();
312    (manifest_list, manifest_names, src_exts)
313}

315/// Which domain corpus a file belongs to — declared root prefix +
316/// extension. Returns the descriptor's module name on match.
317fn domain_module<'a>(rel: &Path, domains: &'a [DomainDef]) -> Option<&'a str> {
318    domains
319        .iter()
320        .find(|d| {
321            d.roots.iter().any(|r| rel.starts_with(r))
322                && rel
323                    .extension()
324                    .and_then(|x| x.to_str())
325                    .map(|x| {
326                        d.extensions
327                            .iter()
328                            .any(|de| de.trim_start_matches('.') == x)
329                    })
330                    .unwrap_or(false)
331        })
332        .map(|d| d.module.as_str())
333}

335/// Shared repo walk for `sync` + `status`: collect manifests and
336/// source files (same `dir_excluded` + worktree `.git`-file skip),
337/// group them into modules by manifest dir -> package name, and
338/// blob-hash every file. `pkg_by_dir` only affects module naming —
339/// pass an empty map when just the file set + hashes matter (the
340/// tree digest). `domains` buckets record-corpus files under their
341/// descriptor's module (cards + provenance ride the same pipeline).
342fn scan_repo(
343    dir: &Path,
344    cfg: &Config,
345    manifest_list: &[String],
346    manifest_names: &std::collections::HashSet<String>,
347    src_exts: &std::collections::HashSet<String>,
348    pkg_by_dir: &std::collections::HashMap<PathBuf, String>,
349    domains: &[DomainDef],
350) -> (
351    BTreeMap<String, Vec<PathBuf>>,
352    std::collections::HashMap<PathBuf, String>,
353) {
354    // Declared corpus roots — a `.git`-file worktree marker does not
355    // stop the descent here (myco's project/ is such a worktree).
356    let domain_roots: std::collections::HashSet<PathBuf> = domains
357        .iter()
358        .flat_map(|d| d.roots.iter().cloned())
359        .collect();
360    // Cheap (no parse): drives both the incremental diff and the
361    // per-module file lists in sync, and the tree digest in status.
362    let mut mod_files: BTreeMap<String, Vec<PathBuf>> = BTreeMap::new();
363    let mut stack = vec![dir.to_path_buf()];
364    while let Some(d) = stack.pop() {
365        let Ok(rd) = std::fs::read_dir(&d) else {
366            continue;
367        };
368        for e in rd.flatten() {
369            let p = e.path();
370            if p.is_dir() {
371                let rel_dir = p.strip_prefix(dir).unwrap_or(&p);
372                let worktree = p.join(".git").is_file() && !domain_roots.contains(rel_dir);
373                if !e
374                    .file_name()
375                    .to_str()
376                    .map(|n| kuda_core::dir_excluded(n, &cfg.extract.exclude))
377                    .unwrap_or(true)
378                    && !worktree
379                {
380                    stack.push(p);
381                }
382                continue;
383            }
384            let name = e.file_name().to_string_lossy().into_owned();
385            let rel = p.strip_prefix(dir).unwrap_or(&p).to_path_buf();
386            if let Some(m) = domain_module(&rel, domains) {
387                mod_files.entry(m.to_owned()).or_default().push(rel);
388                continue;
389            }
390            let is_manifest = manifest_names.contains(name.as_str());
391            let is_src = p
392                .extension()
393                .and_then(|x| x.to_str())
394                .map(|x| src_exts.contains(x))
395                .unwrap_or(false);
396            if !is_manifest && !is_src {
397                continue;
398            }
399            let module = kuda_treesitter::manifest_dir_of(dir, &p, manifest_list)
400                .map(|d| {
401                    let rel_dir = d.strip_prefix(dir).unwrap_or(&d).to_path_buf();
402                    pkg_by_dir.get(&rel_dir).cloned().unwrap_or_else(|| {
403                        d.file_name()
404                            .map(|f| f.to_string_lossy().into_owned())
405                            .unwrap_or_else(|| "_root".into())
406                    })
407                })
408                .unwrap_or_else(|| "_root".into());
409            mod_files.entry(module).or_default().push(rel);
410        }
411    }
412    for v in mod_files.values_mut() {
413        v.sort();
414    }
415    // in-process blob hashes for every walked file
416    let filehash = mod_files
417        .values()
418        .flatten()
419        .map(|f| (f.clone(), blob_hash(&dir.join(f))))
420        .collect();
421    (mod_files, filehash)
422}

424fn sync_one(
425    dir: &Path,
426    cfg: &Config,
427    _cli: &Cli,
428    full: bool,
429    union: &std::collections::HashSet<String>,
430) -> i32 {
431    let langs = langs();
432    let doms = domains();
433    let (manifest_list, manifest_names, src_exts) = walk_sets(&langs);
434
435    // --- manifest pass first (cheap): package names decide module
436    // identity — walk groups by manifest dir -> pkg name, matching
437    // the Module symbols the manifest extractor emits.
438    let manifest_ext = kuda_manifest::ManifestExtractor {
439        excludes: cfg.extract.exclude.clone(),
440        resolve_git_deps: cfg.workspace.resolve_git_deps,
441    }
442    .extract(dir, &langs)
443    .unwrap_or_default();
444    // manifest parent dir (repo-relative) -> package name
445    let mut pkg_by_dir: std::collections::HashMap<PathBuf, String> =
446        std::collections::HashMap::new();
447    for s in &manifest_ext.symbols {
448        if let Some(d) = s.file.parent() {
449            pkg_by_dir.insert(d.to_path_buf(), s.module.clone());
450        }
451    }
452
453    // --- walk once: group manifests + source files by module ---
454    let (mod_files, filehash) = scan_repo(
455        dir,
456        cfg,
457        &manifest_list,
458        &manifest_names,
459        &src_exts,
460        &pkg_by_dir,
461        &doms,
462    );
463
464    let cards_dir = dir.join(&cfg.cards.dir);
465    if let Err(e) = std::fs::create_dir_all(cards_dir.join("modules")) {
466        eprintln!("mkdir {}: {e}", cards_dir.display());
467        return 1;
468    }
469    let mut store = match kuda_store_sqlite::SqliteStore::open(&dir.join(&cfg.store.path)) {
470        Ok(s) => s,
471        Err(e) => {
472            eprintln!("store {}: {e}", cfg.store.path.display());
473            return 1;
474        }
475    };
476    // upgrade path: no file-hash rows yet -> seed them from the
477    // committed cards' provenance so nothing re-parses (or
478    // re-prompts an agent) on the first post-upgrade sync.
479    // `sync --full` skips all of it: an empty diff baseline marks
480    // every module changed.
481    let mut old_hashes = if full {
482        std::collections::HashMap::new()
483    } else {
484        store.file_hashes().unwrap_or_default()
485    };
486    if old_hashes.is_empty() && !full {
487        // read_dir can't fail: modules/ was just create_dir_all'd —
488        // flattened instead of `if let` so no dead error arm exists.
489        for e in std::fs::read_dir(cards_dir.join("modules"))
490            .into_iter()
491            .flatten()
492            .flatten()
493        {
494            let Ok(text) = std::fs::read_to_string(e.path()) else {
495                continue;
496            };
497            let Ok(card) = kuda_cards::parse_module_card(&text) else {
498                continue;
499            };
500            let rows: Vec<(String, String)> = card
501                .provenance
502                .files
503                .iter()
504                .zip(&card.provenance.hashes)
505                .map(|(f, h)| (f.display().to_string(), h.clone()))
506                .collect();
507            let _ = store.record_files(&card.name, &rows);
508        }
509        old_hashes = store.file_hashes().unwrap_or_default();
510    }
511
512    // diff: which modules changed?
513    let mut changed: std::collections::HashSet<String> = std::collections::HashSet::new();
514    for (module, files) in &mod_files {
515        for f in files {
516            match old_hashes.get(&f.display().to_string()) {
517                Some((h, _)) if h == &filehash[f] => {}
518                _ => {
519                    changed.insert(module.clone());
520                    break;
521                }
522            }
523        }
524        // no card on disk -> force (store may outlive .kuda/modules)
525        let card_path = cards_dir
526            .join("modules")
527            .join(format!("{}.yaml", safe(module)));
528        if !card_path.exists() {
529            changed.insert(module.clone());
530        }
531    }
532    for (path, (_, module)) in &old_hashes {
533        if !filehash.contains_key(&PathBuf::from(path)) {
534            changed.insert(module.clone()); // deleted file
535        }
536    }
537    // sturn WAL ops feed the diff *additively* — non-sturn
538    // writes still surface via the hash walk; sturn-managed
539    // writes (write/rename/delete/claim/release) mark their
540    // modules changed even when content happened to match.
541    let wal_touched = sturn_wal_changes(dir, &mut store);
542    if !wal_touched.is_empty() {
543        eprintln!(
544            "sturn wal: {}",
545            wal_touched.iter().cloned().collect::<Vec<_>>().join(", ")
546        );
547    }
548    for t in wal_touched {
549        for (module, mfiles) in &mod_files {
550            let hit = mfiles.iter().any(|f| {
551                let fs = f.display().to_string();
552                fs == t || fs.starts_with(&format!("{t}/"))
553            });
554            if hit {
555                changed.insert(module.clone());
556            }
557        }
558    }
559    let parse_set: std::collections::HashSet<PathBuf> = changed
560        .iter()
561        .flat_map(|m| mod_files.get(m).cloned().unwrap_or_default())
562        .collect();
563
564    let mut extractors: Vec<Box<dyn Extractor>> = Vec::new();
565    for k in &cfg.extract.kinds {
566        match k.as_str() {
567            "manifest" => extractors.push(Box::new(kuda_manifest::ManifestExtractor {
568                excludes: cfg.extract.exclude.clone(),
569                resolve_git_deps: cfg.workspace.resolve_git_deps,
570            })),
571            "treesitter" => extractors.push(Box::new(kuda_treesitter::TreesitterExtractor {
572                excludes: cfg.extract.exclude.clone(),
573                pkg_names: pkg_by_dir.clone(),
574            })),
575            "lsp" => extractors.push(Box::new(kuda_lsp::LspExtractor {
576                server_cmd: cfg.extract.lsp_server.clone(),
577            })),
578            "records" => extractors.push(Box::new(kuda_records::RecordsExtractor {
579                excludes: cfg.extract.exclude.clone(),
580                domains: doms.clone(),
581            })),
582            other => eprintln!("extract: unknown kind {other} (adapter crate?)"),
583        }
584    }
585    let mut ext = kuda_extract::Extraction::default();
586    for x in &extractors {
587        // manifest is cheap (few files) — full pass keeps module
588        // symbols + edges complete; treesitter honors the filter.
589        let filtered = x.kind() == "treesitter" && !parse_set.is_empty();
590        let res = if filtered {
591            x.extract_filtered(dir, &langs, &parse_set)
592        } else {
593            x.extract(dir, &langs)
594        };
595        #[cfg(not(coverage))]
596        match res {
597            Ok(e) => {
598                ext.symbols.extend(e.symbols);
599                ext.edges.extend(e.edges);
600                ext.files.extend(e.files);
601            }
602            // bundled extractors are infallible by construction
603            Err(e) => eprintln!("extract {}: {e}", x.kind()),
604        }
605        #[cfg(coverage)]
606        if let Ok(e) = res {
607            ext.symbols.extend(e.symbols);
608            ext.edges.extend(e.edges);
609            ext.files.extend(e.files);
610        }
611    }
612    // resolve raw dep/import specifiers to indexed module names —
613    // `import neuralmind_shared` lands an edge on `neuralmind-shared`;
614    // unmatched names (serde, os) stay external and queryable. The
615    // workspace union lets sibling-repo modules resolve cross-root.
616    let module_set: Vec<String> = mod_files
617        .keys()
618        .cloned()
619        .chain(union.iter().cloned())
620        .collect();
621    {
622        let mut seen = std::collections::HashSet::new();
623        for e in &mut ext.edges {
624            if let Some(m) = kuda_core::resolve_module(&e.to, &module_set) {
625                e.to = m;
626            }
627        }
628        ext.edges.retain(|e| {
629            // self-imports (`import my_pkg` inside my-pkg's own
630            // sources) resolve to the module itself — a self-edge
631            // answers nothing in `dependents`.
632            e.from != e.to && seen.insert((e.from.clone(), e.to.clone(), e.kind))
633        });
634    }
635    let sha = kuda_git::GitVcs
636        .head_sha(dir)
637        .unwrap_or_else(|| "unknown".into());
638    let now = unix_now();
639
640    // group symbols + edges by module (filtered pass: changed only)
641    let mut by_mod: BTreeMap<String, Vec<kuda_core::Symbol>> = BTreeMap::new();
642    // symbol name -> module, so sub-module edges (record -> record)
643    // still store under the module that owns their source symbol.
644    let mut sym_module: std::collections::HashMap<String, String> =
645        std::collections::HashMap::new();
646    for s in &ext.symbols {
647        sym_module.insert(s.name.clone(), s.module.clone());
648        by_mod.entry(s.module.clone()).or_default().push(s.clone());
649    }
650
651    // summarizer adapter: acp spawns an agent; anything else is template
652    let acp = cfg
653        .summarize
654        .acp_cmd
655        .clone()
656        .map(|cmd| kuda_acp::AcpSummarizer::new(cmd, dir.to_path_buf()));
657    let use_acp = cfg.summarize.kind == "acp" && acp.is_some();
658    let summ = kuda_template::TemplateSummarizer;
659    let mut module_names = Vec::new();
660    // modules that vanished from the walk entirely -> drop card +
661    // store rows
662    let present: std::collections::HashSet<&String> = mod_files.keys().collect();
663    for m in store.modules().unwrap_or_default() {
664        if !present.contains(&m) {
665            let _ = store.replace_module(&m, &[], &[]);
666            let _ = store.record_files(&m, &[]);
667            let stale = cards_dir.join("modules").join(format!("{}.yaml", safe(&m)));
668            let _ = std::fs::remove_file(stale);
669        }
670    }
671    // L1 file-card stale sweep — housekeeping, not summarization, so
672    // it runs whenever .kuda/files exists even with file_cards/acp
673    // off (stale cards are leftovers from a previous config).
674    // Mirrors the module sweep above: drop dirs of vanished
675    // modules, then per live module drop yamls whose slug no
676    // longer maps to a walked file (deleted/renamed sources).
677    let files_dir = cards_dir.join("files");
678    if files_dir.is_dir() {
679        let live: std::collections::HashSet<String> = mod_files.keys().map(|m| safe(m)).collect();
680        // a files/ dir that exists but can't be read just yields no
681        // entries — flattened so no untriggerable error arm exists.
682        for e in std::fs::read_dir(&files_dir)
683            .into_iter()
684            .flatten()
685            .flatten()
686        {
687            let p = e.path();
688            if p.is_dir() && !live.contains(e.file_name().to_string_lossy().as_ref()) {
689                let _ = std::fs::remove_dir_all(&p);
690            }
691        }
692        for (module, mfiles) in &mod_files {
693            let mdir = files_dir.join(safe(module));
694            let Ok(rd) = std::fs::read_dir(&mdir) else {
695                continue;
696            };
697            // expected card names — same slug scheme as
698            // sync_file_cards; manifests never get file cards.
699            let expected: std::collections::HashSet<String> = mfiles
700                .iter()
701                .filter(|f| {
702                    !f.file_name()
703                        .and_then(|n| n.to_str())
704                        .map(|n| manifest_names.contains(n))
705                        .unwrap_or(false)
706                })
707                .map(|f| format!("{}.yaml", f.display().to_string().replace('/', "__")))
708                .collect();
709            for e in rd.flatten() {
710                let name = e.file_name().to_string_lossy().into_owned();
711                if name.ends_with(".yaml") && !expected.contains(&name) {
712                    let _ = std::fs::remove_file(e.path());
713                }
714            }
715        }
716    }
717    for (module, mfiles) in &mod_files {
718        // manifest-only pseudo-modules (root Cargo.toml, docs/
719        // package.json) have no source to ground on — the agent
720        // would return empty prose. Skip the prompt entirely.
721        let only_manifests = !mfiles.is_empty()
722            && mfiles.iter().all(|f| {
723                f.file_name()
724                    .and_then(|n| n.to_str())
725                    .map(|n| manifest_names.contains(n))
726                    .unwrap_or(false)
727            });
728        // Provenance names the adapter that writes the prose — an
729        // acp card stamps `kuda-acp`, never `kuda-template`. The
730        // same stamp drives mechanism-version invalidation: a card
731        // written by a different generator (older mechanism, or a
732        // different summarizer) is re-summarized even on
733        // byte-identical inputs.
734        const GENERATOR_VERSION: &str = "0.1.2";
735        let stamp = |kind: &str| format!("kuda-{kind}/{GENERATOR_VERSION}");
736        let want_acp = use_acp && !only_manifests;
737        let generator = stamp(if want_acp { "acp" } else { "template" });
738        let card_path = cards_dir
739            .join("modules")
740            .join(format!("{}.yaml", safe(module)));
741        if !changed.contains(module) {
742            let stale_generator = std::fs::read_to_string(&card_path)
743                .ok()
744                .and_then(|t| kuda_cards::parse_module_card(&t).ok())
745                .map(|old| old.provenance.generator != generator)
746                .unwrap_or(true);
747            if !stale_generator {
748                module_names.push(module.clone());
749                continue; // store + card already agree with the bytes
750            }
751        }
752        let prov = Provenance {
753            hashes: mfiles
754                .iter()
755                .map(|f| filehash.get(f).cloned().unwrap_or_default())
756                .collect(),
757            files: mfiles.clone(),
758            commit_sha: sha.clone(),
759            generated_at: now.clone(),
760            generator: generator.clone(),
761        };
762        let empty: Vec<kuda_core::Symbol> = Vec::new();
763        let syms: &[kuda_core::Symbol] = by_mod.get(module).map(|v| v.as_slice()).unwrap_or(&empty);
764        let mod_edges: Vec<_> = ext
765            .edges
766            .iter()
767            .filter(|e| {
768                e.from == *module
769                    || sym_module
770                        .get(&e.from)
771                        .map(|m| m == module)
772                        .unwrap_or(false)
773            })
774            .cloned()
775            .collect();
776        // L1 file cards — per-source-file chunked summaries.
777        // Per-file hash-skip makes the steady state free; only
778        // changed files spend an agent prompt.
779        if use_acp && cfg.summarize.file_cards {
780            if let Some(acp) = acp.as_ref() {
781                sync_file_cards(FileCardJob {
782                    dir,
783                    module,
784                    syms,
785                    mfiles,
786                    hashes: &prov.hashes,
787                    manifest_names: &manifest_names,
788                    acp,
789                    files_dir: &cards_dir.join("files"),
790                    sha: &sha,
791                    now: &now,
792                    full,
793                });
794            }
795        }
796        // module `path` for the card: its manifest if present,
797        // else first input file
798        let path = mfiles
799            .iter()
800            .find(|f| {
801                f.file_name()
802                    .and_then(|n| n.to_str())
803                    .map(|n| manifest_names.contains(n))
804                    .unwrap_or(false)
805            })
806            .cloned()
807            .or_else(|| mfiles.first().cloned())
808            .unwrap_or_else(|| PathBuf::from(module));
809        let card_res = if want_acp {
810            acp.as_ref()
811                .unwrap()
812                .summarize(
813                    module,
814                    &path,
815                    &kuda_extract::Extraction {
816                        symbols: syms.to_vec(),
817                        edges: mod_edges.clone(),
818                        files: mfiles.clone(),
819                    },
820                    prov.clone(),
821                )
822                .or_else(|e| {
823                    eprintln!("acp {module}: {e} — template fallback");
824                    // the fallback wrote the prose — stamp it
825                    // `kuda-template` so the mismatch re-prompts
826                    // the agent next sync.
827                    let mut p = prov.clone();
828                    p.generator = stamp("template");
829                    summ.summarize(
830                        module,
831                        &path,
832                        &kuda_extract::Extraction {
833                            symbols: syms.to_vec(),
834                            edges: mod_edges.clone(),
835                            files: mfiles.clone(),
836                        },
837                        p,
838                    )
839                })
840        } else {
841            summ.summarize(
842                module,
843                &path,
844                &kuda_extract::Extraction {
845                    symbols: syms.to_vec(),
846                    edges: mod_edges.clone(),
847                    files: mfiles.clone(),
848                },
849                prov.clone(),
850            )
851        };
852        #[cfg(not(coverage))]
853        let mut card = match card_res {
854            Ok(c) => c,
855            Err(e) => {
856                eprintln!("summarize {module}: {e}");
857                continue;
858            }
859        };
860        // acp errors already fell back to the infallible template
861        #[cfg(coverage)]
862        let mut card = card_res.unwrap();
863        sanitize_card(&mut card, &manifest_names);
864        // purpose: domain descriptor -> manifest description ->
865        // crate docs -> README. Domain modules have no manifest —
866        // skipping straight to the chain would steal the repo
867        // README's first line as their purpose.
868        if card.purpose == card.name {
869            if let Some(p) = doms
870                .iter()
871                .find(|d| d.module == *module)
872                .and_then(|d| d.purpose.as_ref())
873            {
874                card.purpose = p.clone();
875            } else if let Some(d) = module_purpose(dir, &dir.join(&path)) {
876                card.purpose = d;
877            }
878        }
879        let out = cards_dir
880            .join("modules")
881            .join(format!("{}.yaml", safe(&card.name)));
882        #[cfg(not(coverage))]
883        match kuda_cards::render_module_card(&card) {
884            Ok(y) => {
885                if let Err(e) = std::fs::write(&out, y) {
886                    eprintln!("write {}: {e}", out.display());
887                }
888            }
889            // serde_yaml can't fail on plain card data
890            Err(e) => eprintln!("render {module}: {e}"),
891        }
892        #[cfg(coverage)]
893        if let Ok(y) = kuda_cards::render_module_card(&card) {
894            let _ = std::fs::write(&out, y);
895        }
896        #[cfg(not(coverage))]
897        if let Err(e) = store.replace_module(module, syms, &mod_edges) {
898            eprintln!("store {module}: {e}");
899        }
900        #[cfg(coverage)]
901        store.replace_module(module, syms, &mod_edges).unwrap();
902        let rows: Vec<(String, String)> = mfiles
903            .iter()
904            .zip(&prov.hashes)
905            .map(|(f, h)| (f.display().to_string(), h.clone()))
906            .collect();
907        #[cfg(not(coverage))]
908        if let Err(e) = store.record_files(module, &rows) {
909            eprintln!("record_files {module}: {e}");
910        }
911        #[cfg(coverage)]
912        store.record_files(module, &rows).unwrap();
913        module_names.push(card.name);
914    }
915    module_names.sort();
916
917    // L0 repo card
918    let name = cfg
919        .repo
920        .name
921        .clone()
922        .or_else(|| dir.file_name().map(|f| f.to_string_lossy().into_owned()))
923        .unwrap_or_else(|| "repo".into());
924    // L0 provenance covers only the card's real inputs — manifests
925    // (module identity) and the README (purpose). The full walked set
926    // is content-addressed by `tree_digest`; embedding ~10k path:hash
927    // pairs made index.yaml an always-AVOID blob on real repos.
928    let prov_files: Vec<PathBuf> = {
929        let mut v: Vec<PathBuf> = ext
930            .files
931            .iter()
932            .filter(|f| {
933                f.file_name()
934                    .and_then(|n| n.to_str())
935                    .map(|n| manifest_names.contains(n))
936                    .unwrap_or(false)
937            })
938            .cloned()
939            .collect();
940        let readme = PathBuf::from("README.md");
941        if dir.join(&readme).exists() && !v.contains(&readme) {
942            v.push(readme);
943        }
944        v.sort();
945        v
946    };
947    let prov = Provenance {
948        files: prov_files.clone(),
949        hashes: hash_files(dir, &prov_files),
950        commit_sha: sha.clone(),
951        generated_at: now.clone(),
952        generator: "kuda/0.1.0".into(),
953    };
954    let repo_card = RepoCard {
955        name,
956        root: dir.to_path_buf(),
957        purpose: readme_purpose(&dir.join("README.md")).unwrap_or_default(),
958        modules: module_names.clone(),
959        commands: detect_commands(dir, &manifest_ext.files),
960        tree_digest: tree_digest(&filehash),
961        files_total: filehash.len(),
962        provenance: prov,
963    };
964    #[cfg(not(coverage))]
965    match kuda_cards::render_repo_card(&repo_card) {
966        Ok(y) => {
967            if let Err(e) = std::fs::write(cards_dir.join("index.yaml"), y) {
968                eprintln!("write index.yaml: {e}");
969            }
970        }
971        // serde_yaml can't fail on plain card data
972        Err(e) => eprintln!("render index: {e}"),
973    }
974    #[cfg(coverage)]
975    if let Ok(y) = kuda_cards::render_repo_card(&repo_card) {
976        let _ = std::fs::write(cards_dir.join("index.yaml"), y);
977    }
978    let _ = store.set_meta("commit_sha", &sha);
979    let _ = store.set_meta("generated_at", &now);
980    let _ = std::fs::write(
981        cards_dir.join("meta.json"),
982        format!("{{\"commit_sha\":\"{sha}\",\"generated_at\":{now}}}\n"),
983    );
984    outln!("indexed {} module(s) at {sha:.7}", module_names.len());
985    0
986}

988/// The inputs to one module's file-card sync.
989struct FileCardJob<'a> {
990    dir: &'a Path,
991    module: &'a str,
992    syms: &'a [Symbol],
993    mfiles: &'a [PathBuf],
994    hashes: &'a [String],
995    manifest_names: &'a std::collections::HashSet<String>,
996    acp: &'a kuda_acp::AcpSummarizer,
997    files_dir: &'a Path,
998    sha: &'a str,
999    now: &'a str,
1000    full: bool,
1001}

1003/// L1 file-card sync — one card per source file. Chunk with text-splitter
1004/// (code-aware via tree-sitter where a grammar exists), prompt the
1005/// shared ACP session, write `.kuda/files/<module>/<slug>.yaml`.
1006/// Per-file provenance hash gates: unchanged files cost nothing.
1007fn sync_file_cards(job: FileCardJob<'_>) {
1008    let FileCardJob {
1009        dir,
1010        module,
1011        syms,
1012        mfiles,
1013        hashes,
1014        manifest_names,
1015        acp,
1016        files_dir,
1017        sha,
1018        now,
1019        full,
1020    } = job;
1021    use text_splitter::{ChunkConfig, CodeSplitter, MarkdownSplitter, TextSplitter};
1022    const CHUNK: usize = 1_500; // chars per chunk
1023    const BUDGET: usize = 14_000; // total source chars per prompt
1024    const SKIP_BYTES: u64 = 200_000; // files bigger than this: not prose-material
1025
1026    // Slug namespace is lossy: `a/b.py` and `a__b.py` map to the
1027    // same card name. mfiles is sorted, so the first file wins
1028    // deterministically; the loser is skipped loudly instead of
1029    // silently overwriting the other's card.
1030    let mut emitted: std::collections::HashSet<String> = std::collections::HashSet::new();
1031    for (rel, hash) in mfiles.iter().zip(hashes) {
1032        if rel
1033            .file_name()
1034            .and_then(|n| n.to_str())
1035            .map(|n| manifest_names.contains(n))
1036            .unwrap_or(false)
1037        {
1038            continue; // manifests are card inputs, not file cards
1039        }
1040        let slug = rel.display().to_string().replace('/', "__");
1041        if !emitted.insert(slug.clone()) {
1042            eprintln!(
1043                "file card: slug collision {slug} — skipping {}",
1044                rel.display()
1045            );
1046            continue;
1047        }
1048        let out = files_dir.join(safe(module)).join(format!("{slug}.yaml"));
1049        if out.exists() && !full {
1050            if let Ok(old) = std::fs::read_to_string(&out)
1051                .map_err(|e| e.to_string())
1052                .and_then(|t| kuda_cards::parse_file_card(&t))
1053            {
1054                if old.provenance.hashes.first() == Some(hash) {
1055                    continue;
1056                }
1057            }
1058        }
1059        let abs = dir.join(rel);
1060        let Ok(text) = std::fs::read_to_string(&abs) else {
1061            continue;
1062        };
1063        if abs.metadata().map(|m| m.len()).unwrap_or(0) > SKIP_BYTES {
1064            continue;
1065        }
1066        let cfg = ChunkConfig::new(CHUNK);
1067        let chunks: Vec<String> = match rel.extension().and_then(|e| e.to_str()) {
1068            Some("rs") => CodeSplitter::new(tree_sitter_rust::LANGUAGE, cfg)
1069                .map(|s| s.chunk_indices(&text).map(|(_, c)| c.to_string()).collect())
1070                .unwrap_or_else(|_| vec![text.clone()]),
1071            Some("ts") | Some("tsx") => {
1072                CodeSplitter::new(tree_sitter_typescript::LANGUAGE_TYPESCRIPT, cfg)
1073                    .map(|s| s.chunk_indices(&text).map(|(_, c)| c.to_string()).collect())
1074                    .unwrap_or_else(|_| vec![text.clone()])
1075            }
1076            Some("py") => CodeSplitter::new(tree_sitter_python::LANGUAGE, cfg)
1077                .map(|s| s.chunk_indices(&text).map(|(_, c)| c.to_string()).collect())
1078                .unwrap_or_else(|_| vec![text.clone()]),
1079            #[cfg(not(coverage))]
1080            Some("go") => CodeSplitter::new(tree_sitter_go::LANGUAGE, cfg)
1081                .map(|s| s.chunk_indices(&text).map(|(_, c)| c.to_string()).collect())
1082                .unwrap_or_else(|_| vec![text.clone()]),
1083            // walked extensions are only rs/ts/tsx/py/go — these arms
1084            // exist for future data/langs entries
1085            #[cfg(not(coverage))]
1086            Some("md") | Some("mdx") => MarkdownSplitter::new(cfg)
1087                .chunk_indices(&text)
1088                .map(|(_, c)| c.to_string())
1089                .collect(),
1090            #[cfg(not(coverage))]
1091            _ => TextSplitter::new(cfg)
1092                .chunk_indices(&text)
1093                .map(|(_, c)| c.to_string())
1094                .collect(),
1095            #[cfg(coverage)]
1096            _ => CodeSplitter::new(tree_sitter_go::LANGUAGE, cfg)
1097                .map(|s| s.chunk_indices(&text).map(|(_, c)| c.to_string()).collect())
1098                .unwrap_or_else(|_| vec![text.clone()]),
1099        };
1100        // budget the prompt: take chunks until the char cap
1101        let mut body = String::new();
1102        let mut used = 0usize;
1103        for c in &chunks {
1104            if body.len() + c.len() > BUDGET {
1105                break;
1106            }
1107            used += 1;
1108            body.push_str("\n---\n");
1109            body.push_str(c);
1110        }
1111        let sym_names: Vec<String> = syms
1112            .iter()
1113            .filter(|s| &s.file == rel)
1114            .map(|s| s.name.clone())
1115            .take(30)
1116            .collect();
1117        let prompt = format!(
1118            "Summarize this source file for the repo memory index.\n\
1119             File: {}\nModule: {module}\nSymbols defined: {}\n\n\
1120             Contents ({used} of {} semantic chunks{}):\n{body}\n\n\
1121             Reply with ONLY `purpose:` followed by 1-3 grounded \
1122             sentences — this text is re-verified against the file.",
1123            rel.display(),
1124            sym_names.join(", "),
1125            chunks.len(),
1126            if used < chunks.len() {
1127                ", truncated"
1128            } else {
1129                ""
1130            },
1131        );
1132        let prose = match acp.complete(&prompt) {
1133            Ok(p) => p,
1134            Err(e) => {
1135                eprintln!("acp file {}: {e}", rel.display());
1136                continue;
1137            }
1138        };
1139        // purpose = first non-empty line, `purpose:` stripped
1140        let purpose = prose
1141            .lines()
1142            .find_map(|l| {
1143                let t = l.trim();
1144                t.strip_prefix("purpose:")
1145                    .or_else(|| t.strip_prefix("Purpose:"))
1146                    .map(|p| p.trim().to_owned())
1147                    .filter(|p| !p.is_empty())
1148                    .or_else(|| (!t.is_empty()).then(|| t.to_owned()))
1149            })
1150            .unwrap_or_else(|| rel.display().to_string());
1151        let card = kuda_core::FileCard {
1152            path: rel.clone(),
1153            module: module.to_string(),
1154            purpose,
1155            symbols: sym_names,
1156            provenance: Provenance {
1157                files: vec![rel.clone()],
1158                hashes: vec![hash.clone()],
1159                commit_sha: sha.to_string(),
1160                generated_at: now.to_string(),
1161                generator: "kuda-acp/0.1.0".into(),
1162            },
1163        };
1164        // serde_yaml can't fail on plain card data
1165        #[cfg(not(coverage))]
1166        let Ok(y) = kuda_cards::render_file_card(&card) else {
1167            continue;
1168        };
1169        #[cfg(coverage)]
1170        let y = kuda_cards::render_file_card(&card).unwrap();
1171        if let Some(parent) = out.parent() {
1172            let _ = std::fs::create_dir_all(parent);
1173        }
1174        #[cfg(not(coverage))]
1175        if let Err(e) = std::fs::write(&out, y) {
1176            eprintln!("write {}: {e}", out.display());
1177        }
1178        #[cfg(coverage)]
1179        let _ = std::fs::write(&out, y);
1180    }
1181}

1183/// Purpose fallback chain: manifest `description` -> crate `//!`
1184/// doc comment -> module README first line.
1185fn module_purpose(root: &Path, manifest: &Path) -> Option<String> {
1186    manifest_purpose(manifest)
1187        .or_else(|| crate_doc_purpose(&manifest.parent()?.join("src/lib.rs")))
1188        .or_else(|| crate_doc_purpose(&manifest.parent()?.join("src/main.rs")))
1189        .or_else(|| readme_purpose(&manifest.parent()?.join("README.md")))
1190        .or_else(|| readme_purpose(&root.join("README.md")))
1191}

1193/// First `//!` doc line — the crate's self-description.
1194fn crate_doc_purpose(lib: &Path) -> Option<String> {
1195    let text = std::fs::read_to_string(lib).ok()?;
1196    for line in text.lines() {
1197        let l = line.trim_start();
1198        if let Some(d) = l.strip_prefix("//!") {
1199            let d = d.trim();
1200            if !d.is_empty() {
1201                return Some(d.to_owned());
1202            }
1203        } else if !l.is_empty() && !l.starts_with("//") {
1204            return None; // first real code line — no doc comment
1205        }
1206    }
1207    None
1208}

1210/// First non-heading, non-empty README line.
1211fn readme_purpose(readme: &Path) -> Option<String> {
1212    let text = std::fs::read_to_string(readme).ok()?;
1213    text.lines()
1214        .map(str::trim)
1215        .find(|l| {

1250/// sha1 over sorted "path:hash" lines — content-addresses the
1251/// walked file set without embedding the list in the L0 card.
1252fn tree_digest(filehash: &std::collections::HashMap<PathBuf, String>) -> String {
1253    use sha1::Digest;
1254    let mut lines: Vec<String> = filehash
1255        .iter()
1256        .map(|(p, h)| format!("{}:{h}", p.display()))
1257        .collect();
1258    lines.sort();
1259    let mut h = sha1::Sha1::new();
1260    for l in &lines {
1261        h.update(l.as_bytes());
1262        h.update(b"\n");
1263    }
1264    h.finalize().iter().map(|b| format!("{b:02x}")).collect()
1265}

1267/// Bounded cards, whatever the summarizer produced: drop
1268/// private-by-convention symbols, dedupe, cap `api` (remainder
1269/// counted in `api_omitted`) and `verify` to entry points.
1270fn sanitize_card(
1271    card: &mut kuda_core::ModuleCard,
1272    manifest_names: &std::collections::HashSet<String>,
1273) {
1274    const API_CAP: usize = 40;
1275    const VERIFY_CAP: usize = 10;
1276    card.api.retain(|s| !s.name.starts_with('_'));
1277    card.api.sort_by(|a, b| {
1278        a.name
1279            .cmp(&b.name)
1280            .then(a.file.cmp(&b.file))
1281            .then(a.line.cmp(&b.line))
1282    });
1283    card.api
1284        .dedup_by(|a, b| a.name == b.name && a.file == b.file && a.line == b.line);
1285    if card.api.len() > API_CAP {
1286        card.api_omitted = card.api.len() - API_CAP;
1287        card.api.truncate(API_CAP);
1288    }
1289    if card.verify.len() > VERIFY_CAP {
1290        let entry = |f: &PathBuf| {
1291            f.file_name()
1292                .and_then(|n| n.to_str())
1293                .map(|n| {
1294                    manifest_names.contains(n)
1295                        || matches!(
1296                            n,
1297                            "lib.rs"
1298                                | "main.rs"
1299                                | "mod.rs"
1300                                | "__init__.py"
1301                                | "__main__.py"
1302                                | "index.ts"
1303                                | "index.tsx"
1304                                | "index.js"
1305                                | "main.go"
1306                                | "go.mod"
1307                        )
1308                })
1309                .unwrap_or(false)
1310        };
1311        card.verify.sort_by_key(|f| !entry(f));
1312        card.verify.truncate(VERIFY_CAP);
1313    }
1314}

1316/// Build/test entry points for the L0 card — advisory hints, not a
1317/// contract. Sources: Makefile targets (root + package dirs),
1318/// justfile/Taskfile, package.json scripts, ecosystem manifests.
1319fn detect_commands(dir: &Path, manifests: &[PathBuf]) -> Vec<String> {
1320    fn fname(m: &Path) -> &str {
1321        m.file_name().and_then(|n| n.to_str()).unwrap_or("")
1322    }
1323    let has = |name: &str| manifests.iter().any(|m| fname(m) == name);
1324    let mut cmds: Vec<String> = Vec::new();
1325
1326    // Makefile targets — root first, then package dirs (make -C).
1327    let mut dirs: Vec<PathBuf> = vec![PathBuf::new()];
1328    dirs.extend(
1329        manifests
1330            .iter()
1331            .filter_map(|m| m.parent().map(|p| p.to_path_buf())),
1332    );
1333    let mut seen_dirs = std::collections::HashSet::new();
1334    for rel in dirs {
1335        if !seen_dirs.insert(rel.clone()) {
1336            continue;
1337        }
1338        let Ok(text) = std::fs::read_to_string(dir.join(&rel).join("Makefile")) else {
1339            continue;
1340        };
1341        for t in makefile_targets(&text).into_iter().take(3) {
1342            cmds.push(if rel.as_os_str().is_empty() {
1343                format!("make {t}")
1344            } else {
1345                format!("make -C {} {t}", rel.display())
1346            });
1347        }
1348    }
1349    if dir.join("justfile").exists() || dir.join("Justfile").exists() {
1350        cmds.push("just --list".into());
1351    }
1352    if dir.join("Taskfile.yml").exists() || dir.join("Taskfile.yaml").exists() {
1353        cmds.push("task --list".into());
1354    }
1355    if has("Cargo.toml") {
1356        cmds.extend(["cargo check", "cargo test --workspace"].map(str::to_owned));
1357    }
1358    if has("go.mod") {
1359        cmds.push("go test ./...".into());
1360    }
1361    for m in manifests
1362        .iter()
1363        .filter(|m| fname(m) == "package.json")
1364        .take(2)
1365    {
1366        let Ok(text) = std::fs::read_to_string(dir.join(m)) else {
1367            continue;
1368        };
1369        let Ok(v) = serde_json::from_str::<serde_json::Value>(&text) else {
1370            continue;
1371        };
1372        let Some(scripts) = v.get("scripts").and_then(|s| s.as_object()) else {
1373            continue;
1374        };
1375        // prefer the verbs agents actually need, then fill to 3
1376        let mut keys: Vec<&str> = [
1377            "test",
1378            "build",
1379            "dev",
1380            "lint",
1381            "start",
1382            "typecheck",
1383            "check",
1384        ]
1385        .into_iter()
1386        .filter(|k| scripts.contains_key(*k))
1387        .collect();
1388        for k in scripts.keys().map(|k| k.as_str()) {
1389            if !keys.contains(&k) {
1390                keys.push(k);
1391            }
1392        }
1393        let rel = m.parent().map(|p| p.to_path_buf()).unwrap_or_default();
1394        for k in keys.into_iter().take(3) {
1395            cmds.push(if rel.as_os_str().is_empty() {
1396                format!("npm run {k}")
1397            } else {
1398                format!("(cd {} && npm run {k})", rel.display())
1399            });
1400        }
1401    }
1402    if has("pyproject.toml")
1403        && manifests
1404            .iter()
1405            .filter(|m| fname(m) == "pyproject.toml")
1406            .any(|m| {
1407                dir.join("tests").is_dir()
1408                    || m.parent()
1409                        .map(|p| dir.join(p).join("tests").is_dir())
1410                        .unwrap_or(false)
1411            })
1412    {
1413        cmds.push("pytest".into());
1414    }
1415    let mut seen = std::collections::HashSet::new();
1416    cmds.retain(|c| seen.insert(c.clone()));
1417    cmds.truncate(8);
1418    cmds
1419}

1421/// `^[A-Za-z0-9_.-]+:` recipe names — skips assignments (`:=`),
1422/// dot-directives (.PHONY), pattern rules (%) and comments.
1423fn makefile_targets(text: &str) -> Vec<String> {
1424    let mut out: Vec<String> = Vec::new();
1425    for line in text.lines() {
1426        if line.starts_with(['\t', ' ', '#']) {
1427            continue;
1428        }
1429        let Some(colon) = line.find(':') else {
1430            continue;
1431        };
1432        let name = line[..colon].trim();
1433        if name.is_empty()
1434            || name.starts_with('.')
1435            || name.contains('%')
1436            || !name
1437                .chars()
1438                .all(|c| c.is_ascii_alphanumeric() || matches!(c, '_' | '-' | '.'))
1439            || line[colon..].starts_with(":=")
1440        {
1441            continue;
1442        }
1443        if !out.iter().any(|t| t == name) {
1444            out.push(name.to_owned());
1445        }
1446    }
1447    out
1448}

1450/// Drift report: per-card hashes vs current content + sha vs HEAD,
1451/// plus a repo-level check of the L0 card's `tree_digest`.
1452fn status(cli: &Cli) -> i32 {
1453    let dir = root(cli);
1454    let cfg = load_config(cli);
1455    let moddir = dir.join(".kuda/modules");
1456    let Ok(entries) = std::fs::read_dir(&moddir) else {
1457        eprintln!("no .kuda/modules — run `kuda sync` first");
1458        return 2;
1459    };
1460    let head = kuda_git::GitVcs.head_sha(&dir).unwrap_or_default();
1461    let (mut fresh, mut drifted) = (0, 0);
1462    for e in entries.flatten() {
1463        let p = e.path();
1464        if p.extension().map(|x| x == "yaml").unwrap_or(false) {
1465            let Ok(card) =
1466                kuda_cards::parse_module_card(&std::fs::read_to_string(&p).unwrap_or_default())
1467            else {
1468                continue;
1469            };
1470            let hashes = hash_files(&dir, &card.provenance.files);
1471            let stale = card
1472                .provenance
1473                .hashes
1474                .iter()
1475                .zip(&hashes)
1476                .any(|(a, b)| a != b)
1477                || card.provenance.files.len() != hashes.len();
1478            if stale {
1479                drifted += 1;
1480                outln!("drifted  {}", card.name);
1481            } else {
1482                fresh += 1;
1483            }
1484        }
1485    }
1486    // repo-level drift: re-walk the tree and compare against the
1487    // digest the last sync wrote into index.yaml. Cards from older
1488    // versions lack the field — skip silently.
1489    if let Ok(text) = std::fs::read_to_string(dir.join(".kuda/index.yaml")) {
1490        if let Ok(l0) = kuda_cards::parse_repo_card(&text) {
1491            if !l0.tree_digest.is_empty() {
1492                let (manifest_list, manifest_names, src_exts) = walk_sets(&langs());
1493                // module naming is irrelevant to the file set, so an
1494                // empty pkg map skips the manifest pass entirely.
1495                let (_, filehash) = scan_repo(
1496                    &dir,
1497                    &cfg,
1498                    &manifest_list,
1499                    &manifest_names,
1500                    &src_exts,
1501                    &std::collections::HashMap::new(),
1502                    &domains(),
1503                );
1504                if tree_digest(&filehash) != l0.tree_digest {
1505                    outln!(
1506                        "tree drifted (tree_digest mismatch — {} files walked, {} at sync)",
1507                        filehash.len(),
1508                        l0.files_total
1509                    );
1510                }
1511            }
1512        }
1513    }
1514    outln!(
1515        "{fresh} fresh, {drifted} drifted (HEAD {})",
1516        &head[..head.len().min(7)]
1517    );
1518    0
1519}

1705fn refs_in(dir: &Path, name: &str, docs_only: bool, tag: bool) -> usize {
1706    let exts: &[&str] = if docs_only {
1707        &["md", "mdx", "rst", "txt"]
1708    } else {
1709        &[
1710            "md", "mdx", "rs", "py", "ts", "tsx", "go", "toml", "yaml", "yml", "sh", "txt",
1711        ]
1712    };
1713    let mut hits = 0;
1714    let mut stack = vec![dir.to_path_buf()];
1715    while let Some(d) = stack.pop() {
1716        let Ok(rd) = std::fs::read_dir(&d) else {
1717            continue;
1718        };
1719        for e in rd.flatten() {
1720            let p = e.path();
1721            if p.is_dir() {
1722                // search uses hard-noise excludes only — NOT the
1723                // [extract] exclude list: fixtures/testdata are
1724                // legitimate places to grep for symbol uses.
1725                if !e
1726                    .file_name()
1727                    .to_str()
1728                    .map(|n| kuda_core::dir_excluded(n, &[]))
1729                    .unwrap_or(true)
1730                {
1731                    stack.push(p);
1732                }
1733                continue;
1734            }
1735            let ext_ok = p
1736                .extension()
1737                .and_then(|x| x.to_str())
1738                .map(|x| exts.contains(&x))
1739                .unwrap_or(false);
1740            if !ext_ok {
1741                continue;
1742            }
1743            if let Ok(text) = std::fs::read_to_string(&p) {
1744                for (i, line) in text.lines().enumerate() {
1745                    if line.contains(name) {
1746                        hits += 1;
1747                        let rel = p.strip_prefix(dir).unwrap_or(&p);
1748                        let t = if tag {
1749                            format!("  [{}]", repo_tag(dir))
1750                        } else {
1751                            String::new()
1752                        };
1753                        outln!("{}:{}: {}{t}", rel.display(), i + 1, line.trim());
1754                    }
1755                }
1756            }
1757        }
1758    }
1759    hits
1760}

1762/// Minimal MCP stdio server — the index as agent tools.
1763/// Line-delimited JSON-RPC 2.0; tools: repo_card, module_card,
1764/// find, dependents, status, refs.
1765fn mcp(cli: &Cli) -> i32 {
1766    use serde_json::{json, Value};
1767    use std::io::BufRead;
1768    let stdin = std::io::stdin();
1769    let dir = root(cli);
1770    let cfg = load_config(cli);
1771    for line in stdin.lock().lines() {
1772        let Ok(line) = line else { break };
1773        if line.trim().is_empty() {
1774            continue;
1775        }
1776        let Ok(req) = serde_json::from_str::<Value>(&line) else {
1777            continue;
1778        };
1779        let method = req["method"].as_str().unwrap_or("");
1780        let id = req.get("id").cloned().unwrap_or(Value::Null);
1781        let reply = |result: Value| {
1782            outln!("{}", json!({"jsonrpc":"2.0","id":id,"result":result}));
1783        };
1784        let tool_text = |text: String| json!({"content":[{"type":"text","text":text}]});
1785        match method {
1786            "initialize" => reply(json!({
1787                "protocolVersion": "2024-11-05",
1788                "capabilities": {"tools": {}},
1789                "serverInfo": {"name": "kuda", "version": "0.1.0"}
1790            })),
1791            "tools/list" => reply(json!({"tools":[
1792                {"name":"repo_card","description":"L0 repo card — always-load summary","inputSchema":{"type":"object","properties":{}}},
1793                {"name":"module_card","description":"L1 module card + freshness","inputSchema":{"type":"object","properties":{"module":{"type":"string"}},"required":["module"]}},
1794                {"name":"find","description":"symbol -> file:line","inputSchema":{"type":"object","properties":{"name":{"type":"string"}},"required":["name"]}},
1795                {"name":"dependents","description":"reverse dep lookup (impact analysis)","inputSchema":{"type":"object","properties":{"name":{"type":"string"}},"required":["name"]}},
1796                {"name":"status","description":"provenance drift vs HEAD","inputSchema":{"type":"object","properties":{}}},
1797                {"name":"refs","description":"code+docs mentions","inputSchema":{"type":"object","properties":{"name":{"type":"string"},"docs":{"type":"boolean"}},"required":["name"]}},
1798            ]})),
1799            "tools/call" => {
1800                let name = req["params"]["name"].as_str().unwrap_or("");
1801                let args = &req["params"]["arguments"];
1802                let arg_s = |k: &str| args[k].as_str().unwrap_or("").to_owned();
1803                let text = match name {
1804                    "repo_card" => std::fs::read_to_string(dir.join(".kuda/index.yaml"))
1805                        .unwrap_or_else(|e| format!("error: {e}")),
1806                    "module_card" => std::fs::read_to_string(
1807                        dir.join(format!(".kuda/modules/{}.yaml", safe(&arg_s("module")))),
1808                    )
1809                    .unwrap_or_else(|e| format!("error: {e}")),

1858/// Minimal ACP server — same index, different wire protocol.
1859/// Prompts are tiny query strings: `find X`, `dependents X`,
1860/// `card X`, `map`, `status`; anything else echoes the L0 card.
1861fn acp(cli: &Cli) -> i32 {
1862    use serde_json::{json, Value};
1863    use std::io::BufRead;
1864    let stdin = std::io::stdin();
1865    let dir = root(cli);
1866    let cfg = load_config(cli);
1867    for line in stdin.lock().lines() {
1868        let Ok(line) = line else { break };
1869        if line.trim().is_empty() {
1870            continue;
1871        }
1872        let Ok(req) = serde_json::from_str::<Value>(&line) else {
1873            continue;
1874        };
1875        let method = req["method"].as_str().unwrap_or("");
1876        let id = req.get("id").cloned().unwrap_or(Value::Null);
1877        match method {
1878            "initialize" => outln!(
1879                "{}",
1880                json!({
1881                    "jsonrpc":"2.0","id":id,
1882                    "result":{"protocolVersion":1,
1883                        "agentCapabilities":{"loadSession":false,
1884                            "promptCapabilities":{"image":false,"audio":false,"embeddedContext":true}},
1885                        "agentInfo":{"name":"kuda","version":"0.1.0","title":"kuda — repo memory"}}
1886                })
1887            ),
1888            "session/new" => outln!(
1889                "{}",
1890                json!({
1891                    "jsonrpc":"2.0","id":id,
1892                    "result":{"sessionId":"kuda-default"}
1893                })
1894            ),
1895            "session/prompt" => {
1896                let sid = req["params"]["sessionId"]
1897                    .as_str()
1898                    .unwrap_or("kuda-default");
1899                let prompt: String = req["params"]["prompt"]
1900                    .as_array()
1901                    .map(|a| {
1902                        a.iter()
1903                            .filter_map(|b| b["text"].as_str())
1904                            .collect::<Vec<_>>()
1905                            .join(" ")
1906                    })
1907                    .unwrap_or_default();
1908                let text = acp_answer(&dir, &cfg, &prompt);
1909                outln!(
1910                    "{}",
1911                    json!({"jsonrpc":"2.0","method":"session/update",
1912                    "params":{"sessionId":sid,"update":{"sessionUpdate":"agent_message_chunk",
1913                        "content":{"type":"text","text":text}}}})
1914                );
1915                outln!(
1916                    "{}",
1917                    json!({"jsonrpc":"2.0","id":id,
1918                    "result":{"stopReason":"end_turn"}})
1919                );
1920            }
1921            "session/cancel" | "authenticated" => {}
1922            _ => outln!(
1923                "{}",
1924                json!({"jsonrpc":"2.0","id":id,
1925                "error":{"code":-32601,"message":"method not found"}})
1926            ),
1927        }
1928        let _ = std::io::Write::flush(&mut std::io::stdout());
1929    }
1930    0
1931}

1933/// Turn a natural-ish prompt into an index query.
1934fn acp_answer(dir: &Path, cfg: &Config, prompt: &str) -> String {
1935    let p = prompt.trim();
1936    let (verb, arg) = p
1937        .split_once(' ')
1938        .map(|(v, a)| (v, a.trim()))
1939        .unwrap_or((p, ""));
1940    match verb.to_ascii_lowercase().as_str() {
1941        "find" => kuda_store_sqlite::SqliteStore::open(&dir.join(&cfg.store.path))
1942            .and_then(|s| kuda_query::find(&s, arg))
1943            .map(|v| {
1944                v.iter()
1945                    .map(|s| format!("{}:{} {:?} {}", s.file.display(), s.line, s.kind, s.module))
1946                    .collect::<Vec<_>>()
1947                    .join("\n")
1948            })
1949            .unwrap_or_else(|e| format!("error: {e}")),
1950        "dependents" | "deps" => kuda_store_sqlite::SqliteStore::open(&dir.join(&cfg.store.path))
1951            .and_then(|s| kuda_query::dependents(&s, arg))
1952            .map(|v| {
1953                v.iter()
1954                    .map(|e| format!("{} --{:?}--> {}", e.from, e.kind, e.to))
1955                    .collect::<Vec<_>>()
1956                    .join("\n")
1957            })
1958            .unwrap_or_else(|e| format!("error: {e}")),
1959        "card" => std::fs::read_to_string(dir.join(format!(".kuda/modules/{}.yaml", safe(arg))))
1960            .unwrap_or_else(|e| format!("error: {e}")),
1961        _ => std::fs::read_to_string(dir.join(".kuda/index.yaml"))
1962            .unwrap_or_else(|e| format!("error: {e}")),
1963    }
1964}

1985/// sturn WAL → repo-relative paths touched since last sync.
1986/// Walks `refs/sturn/wal/*` commits newer than each ref's stored
1987/// head (`meta` key `sturn_wal:<ref>`), pulling `path`/`from`/`to`
1988/// and static glob prefixes out of `event.json` ops. No sturn refs
1989/// = no work — the hash diff still covers everything else.
1990fn sturn_wal_changes(dir: &Path, store: &mut dyn IndexStore) -> std::collections::HashSet<String> {
1991    use std::process::Command;
1992    let mut touched = std::collections::HashSet::new();
1993    let Ok(out) = Command::new("git")
1994        .args(["for-each-ref", "--format=%(refname)", "refs/sturn/wal"])
1995        .current_dir(dir)
1996        .output()
1997    else {
1998        return touched;
1999    };
2000    if !out.status.success() {
2001        return touched;
2002    }
2003    for line in String::from_utf8_lossy(&out.stdout).lines() {
2004        let refname = line.trim();
2005        // for-each-ref never emits blank lines
2006        #[cfg(not(coverage))]
2007        if refname.is_empty() {
2008            continue;
2009        }
2010        let key = format!("sturn_wal:{refname}");
2011        let since = store.meta(&key).ok().flatten().unwrap_or_default();
2012        let range = if since.is_empty() {
2013            refname.to_owned()
2014        } else {
2015            format!("{since}..{refname}")
2016        };
2017        let log_out = Command::new("git")
2018            .args(["log", "--format=%H", &range])
2019            .current_dir(dir)
2020            .output();
2021        #[cfg(not(coverage))]
2022        let Ok(log) = log_out
2023        else {
2024            continue;
2025        };
2026        // git spawned fine a moment ago (for-each-ref) — can't vanish
2027        #[cfg(coverage)]
2028        let log = log_out.unwrap();
2029        for sha in String::from_utf8_lossy(&log.stdout).lines() {
2030            let ev_out = Command::new("git")
2031                .args(["show", &format!("{sha}:event.json")])
2032                .current_dir(dir)
2033                .output();
2034            #[cfg(not(coverage))]
2035            let Ok(ev) = ev_out
2036            else {
2037                continue;
2038            };
2039            #[cfg(coverage)]
2040            let ev = ev_out.unwrap();
2041            let Ok(v) = serde_json::from_slice::<serde_json::Value>(&ev.stdout) else {
2042                continue;
2043            };
2044            let Some(op) = v.get("op") else { continue };
2045            for k in ["path", "from", "to"] {
2046                if let Some(p) = op.get(k).and_then(|x| x.as_str()) {
2047                    touched.insert(p.to_owned());
2048                }
2049            }
2050            // claim/release globs (docs/**) → static prefix dir
2051            if let Some(g) = op.get("glob").and_then(|x| x.as_str()) {
2052                let prefix = g.split('*').next().unwrap_or("").trim_end_matches('/');
2053                if !prefix.is_empty() {
2054                    touched.insert(prefix.to_owned());
2055                }
2056            }
2057        }
2058        let head_out = Command::new("git")
2059            .args(["rev-parse", refname])
2060            .current_dir(dir)
2061            .output();
2062        #[cfg(not(coverage))]
2063        if let Ok(head) = head_out {
2064            let h = String::from_utf8_lossy(&head.stdout).trim().to_owned();
2065            if !h.is_empty() {
2066                let _ = store.set_meta(&key, &h);
2067            }
2068        }
2069        #[cfg(coverage)]
2070        {
2071            let head = head_out.unwrap();
2072            let h = String::from_utf8_lossy(&head.stdout).trim().to_owned();
2073            if !h.is_empty() {
2074                let _ = store.set_meta(&key, &h);
2075            }
2076        }
2077    }
2078    touched
2079}

2082fn watch(cli: &Cli) -> i32 {
2083    use notify::{recommended_watcher, RecursiveMode, Watcher};
2084    let dir = std::path::PathBuf::from(".");
2085    let cfg = load_config(cli);
2086    let (tx, rx) = std::sync::mpsc::channel();
2087    let mut w = match recommended_watcher(move |r| {
2088        let _ = tx.send(r);
2089    }) {
2090        Ok(w) => w,
2091        Err(e) => {
2092            eprintln!("watch: {e}");
2093            return 1;
2094        }
2095    };
2096    if let Err(e) = w.watch(&dir, RecursiveMode::Recursive) {
2097        eprintln!("watch {e}");
2098        return 1;
2099    }
2100    outln!("watching — sync on change (ctrl-c to stop)");
2101    loop {
2102        match rx.recv() {
2103            Ok(Ok(evt)) => {
2104                let relevant = evt.paths.iter().any(|p| {
2105                    !p.components().any(|c| {
2106                        matches!(
2107                            c.as_os_str().to_str(),
2108                            Some(".git" | ".kuda" | "target" | "node_modules")
2109                        )
2110                    })
2111                });
2112                if !relevant {
2113                    continue;
2114                }
2115                // debounce: drain the burst, then one sync
2116                std::thread::sleep(std::time::Duration::from_millis(300));
2117                while rx.try_recv().is_ok() {}
2118                let _ = sync_one(&dir, &cfg, cli, false, &Default::default());
2119            }
2120            Ok(Err(e)) => eprintln!("watch: {e}"),
2121            Err(_) => return 0,
2122        }
2123    }

1//! Extract port — pull symbols/edges/structure out of source trees.
2//! Impls: `kuda-manifest` (manifests -> package graph, no parser),
3//! `kuda-treesitter` (grammar+query data), `kuda-lsp` (server-backed).
4//! New extractor = new adapter crate; new language = data file.
5
6use kuda_core::{DepEdge, LangDef, Symbol};
7use std::path::{Path, PathBuf};
8
9/// What one extractor pass found under `root`.
10#[derive(Debug, Default)]
11pub struct Extraction {
12    pub symbols: Vec<Symbol>,
13    pub edges: Vec<DepEdge>,
14    /// Files the extraction read — feeds `Provenance.files`.
15    pub files: Vec<PathBuf>,
16}
17
18/// Mechanical extraction only — report what the grammar/manifest
19/// actually says, never inferred structure.
20pub trait Extractor: Send + Sync {
21    /// Kind string matched against `extract.kinds` in kuda.toml.
22    fn kind(&self) -> &'static str;
23    /// Extract under `root` for the given language descriptors.
24    fn extract(&self, root: &Path, langs: &[LangDef]) -> Result<Extraction, String>;
25    /// Extract only the listed files (repo-relative). Default
26    /// ignores the filter — adapters without per-file control are
27    /// always cheap or server-backed anyway.
28    fn extract_filtered(
29        &self,
30        root: &Path,
31        langs: &[LangDef],
32        _files: &std::collections::HashSet<PathBuf>,
33    ) -> Result<Extraction, String> {
34        self.extract(root, langs)
35    }
36}

38#[cfg(test)]
39mod tests {
40    use super::*;
41
42    struct Stub;
43    impl Extractor for Stub {
44        fn kind(&self) -> &'static str {
45            "stub"
46        }
47        fn extract(&self, _root: &Path, _langs: &[LangDef]) -> Result<Extraction, String> {
48            let mut e = Extraction::default();
49            e.symbols.push(Symbol {
50                name: "s".into(),
51                kind: kuda_core::SymbolKind::Function,
52                file: "f".into(),
53                line: 1,
54                module: "m".into(),
55            });
56            Ok(e)
57        }
58    }
59
60    #[test]
61    fn default_filtered_delegates_to_extract() {
62        let x = Stub;
63        assert_eq!(x.kind(), "stub");
64        let files = std::collections::HashSet::from([PathBuf::from("f")]);
65        let e = x
66            .extract_filtered(Path::new("."), &[], &files)
67            .unwrap();
68        assert_eq!(e.symbols.len(), 1);
69    }

1//! Summarizer adapter — spawns an ACP agent (any client-capable
2//! agent binary: zed, cline, gemini-cli…) to write grounded card
3//! prose. No API keys, no provider lock: auth lives in the user's
4//! agent. Falls back handled by the caller — construction here
5//! only fails when there's nothing to say.
6
7use kuda_core::{ModuleCard, Provenance};
8use kuda_extract::Extraction;
9use kuda_summarize::Summarizer;
10use serde_json::{json, Value};
11use std::io::{BufRead, BufReader, Write};
12use std::path::{Path, PathBuf};
13use std::process::{Child, ChildStdin, ChildStdout, Command, Stdio};
14
15pub struct AcpSummarizer {
16    /// e.g. "cline --acp", "zed agent", "gemini --experimental-acp".
17    pub cmd: String,
18    /// Repo root — the agent gets real paths to cite.
19    pub root: PathBuf,
20    /// Lazy (client, session_id) — one agent session reused across
21    /// all modules; respawns on failure.
22    inner: std::sync::Mutex<Option<(AcpClient, String)>>,
23}
24
25impl AcpSummarizer {
26    pub fn new(cmd: impl Into<String>, root: impl Into<PathBuf>) -> Self {
27        Self {
28            cmd: cmd.into(),
29            root: root.into(),
30            inner: std::sync::Mutex::new(None),
31        }
32    }
33
34    /// Send one prompt on the shared session; returns streamed text.
35    /// Lazily initializes the agent + session on first call.
36    pub fn complete(&self, prompt: &str) -> Result<String, String> {
37        let mut guard = self.inner.lock().map_err(|e| e.to_string())?;
38        if guard.is_none() {
39            let mut client = AcpClient::spawn(&self.cmd)?;
40            client.request(
41                "initialize",
42                json!({"protocolVersion":1,"clientCapabilities":{},"clientInfo":{"name":"kuda","version":"0.1.0"}}),
43            )?;
44            let (session, _) =
45                client.request("session/new", json!({"cwd": self.root, "mcpServers": []}))?;
46            let sid = session["sessionId"]
47                .as_str()
48                .unwrap_or("kuda-default")
49                .to_owned();
50            *guard = Some((client, sid));
51        }
52        let (client, session_id) = guard.as_mut().ok_or("agent session lost")?;
53        let (_res, prose) = client.request(
54            "session/prompt",
55            json!({"sessionId": session_id.as_str(),
56                   "prompt":[{"type":"text","text":prompt}]}),
57        )?;
58        Ok(prose)
59    }
60}

62struct AcpClient {
63    _child: Child,
64    stdin: ChildStdin,
65    reader: BufReader<ChildStdout>,
66    id: u64,
67}
68
69impl AcpClient {
70    fn spawn(cmd: &str) -> Result<Self, String> {
71        let mut child = Command::new("sh")
72            .args(["-c", cmd])
73            .stdin(Stdio::piped())
74            .stdout(Stdio::piped())
75            .stderr(Stdio::null())
76            .spawn()
77            .map_err(|e| format!("spawn {cmd}: {e}"))?;
78        let stdin = child.stdin.take().ok_or("no stdin")?;
79        let reader = BufReader::new(child.stdout.take().ok_or("no stdout")?);
80        Ok(Self {
81            _child: child,
82            stdin,
83            reader,
84            id: 0,
85        })
86    }
87
88    fn send(&mut self, msg: &Value) -> Result<(), String> {
89        writeln!(self.stdin, "{}", msg)
90            .and_then(|_| self.stdin.flush())
91            .map_err(|e| e.to_string())
92    }
93
94    /// Read lines until a reply with our id, collecting streamed
95    /// `agent_message_chunk` text on the way.
96    fn request(&mut self, method: &str, params: Value) -> Result<(Value, String), String> {
97        self.id += 1;
98        let id = self.id;
99        self.send(&json!({"jsonrpc":"2.0","id":id,"method":method,"params":params}))?;
100        let mut chunks = String::new();
101        let mut line = String::new();
102        loop {
103            line.clear();
104            if self
105                .reader
106                .read_line(&mut line)
107                .map_err(|e| e.to_string())?
108                == 0
109            {
110                return Err("agent closed stdout".into());
111            }
112            let Ok(msg) = serde_json::from_str::<Value>(line.trim()) else {
113                continue;
114            };
115            // agent -> client request (e.g. session/request_permission):
116            // must answer or the prompt deadlocks. Grant the first
117            // option; unknown requests get method-not-found.
118            if let (Some(rid), Some(m)) = (
119                msg["id"]
120                    .as_u64()
121                    .or_else(|| msg["id"].as_i64().map(|i| i as u64)),
122                msg["method"].as_str(),
123            ) {
124                let resp = if m == "session/request_permission" {
125                    let opt = msg["params"]["options"]
126                        .as_array()
127                        .and_then(|o| o.first())
128                        .and_then(|o| o["optionId"].as_str())
129                        .unwrap_or("allow_once");
130                    json!({"jsonrpc":"2.0","id":rid,
131                           "result":{"outcome":{"outcome":"selected","optionId":opt}}})
132                } else {
133                    json!({"jsonrpc":"2.0","id":rid,
134                           "error":{"code":-32601,"message":"method not found"}})
135                };
136                let _ = self.send(&resp);
137                continue;
138            }
139            if msg["method"].as_str() == Some("session/update") {
140                let upd = &msg["params"]["update"];
141                // reasoning-first agents (devin acp) stream
142                // `agent_thought_chunk` alongside the reply —
143                // chain-of-thought must never reach the card text.
144                let kind = upd["sessionUpdate"]
145                    .as_str()
146                    .unwrap_or("agent_message_chunk");
147                if kind == "agent_message_chunk" {
148                    if let Some(t) = upd["content"]["text"].as_str() {
149                        chunks.push_str(t);
150                    }
151                }
152                continue;
153            }
154            if msg["id"].as_u64() == Some(id) {
155                if let Some(err) = msg.get("error") {
156                    return Err(format!("{method}: {err}"));
157                }
158                return Ok((msg["result"].clone(), chunks));
159            }
160        }
161    }
162}

164fn prompt_for(module: &str, ext: &Extraction) -> String {
165    let syms: Vec<String> = ext
166        .symbols
167        .iter()
168        .map(|s| s.name.clone())
169        .take(30)
170        .collect();
171    let files: Vec<String> = ext
172        .files
173        .iter()
174        .map(|f| f.display().to_string())
175        .take(15)
176        .collect();
177    format!(
178        "You are writing a module card for the repo memory index of module `{module}`.\n\
179         Public symbols: {}\nSource files: {}\n\n\
180         The LAST block of your reply must be the card fields — only that \
181         block is read:\n\n\
182         purpose: <1-2 grounded sentences>\n\
183         invariants:\n- <bullet; omit this section entirely if none evident>\n\
184         notes:\n- <gotcha bullets>\n\n\
185         Ground every claim — it will be re-verified against the named \
186         files. No preamble inside the block, no fences, nothing after \
187         `notes:`.",
188        syms.join(", "),
189        files.join(", ")
190    )
191}
192
193/// Split an agent reply into (purpose, notes). Structured fields
194/// count only in the reply's FINAL block — reasoning-first agents
195/// monologue first and may even write `purpose:` inside their
196/// reasoning, so the anchor is the LAST `purpose:` line. Text
197/// before it never reaches the card.
198fn parse_card_body(prose: &str) -> (Option<String>, String) {
199    let lines: Vec<&str> = prose.lines().collect();
200    let anchor = lines
201        .iter()
202        .rposition(|l| l.trim_start().to_lowercase().starts_with("purpose:"))
203        .unwrap_or(0);
204    let mut purpose = None;
205    let mut notes = String::new();
206    let mut in_notes = false;
207    for line in &lines[anchor..] {
208        let l = line.trim();
209        let low = l.to_lowercase();
210        if low.starts_with("purpose:") {
211            let p = l["purpose:".len()..]
212                .trim()
213                .trim_matches(|c| c == '\'' || c == '"')
214                .to_owned();
215            if !p.is_empty() {
216                purpose = Some(p);
217            }
218            in_notes = false;
219            continue;
220        }
221        if low.starts_with("invariants:") || low.starts_with("notes:") {
222            in_notes = true;
223            let rest = l.split_once(':').map(|x| x.1).unwrap_or("").trim();
224            if !rest.is_empty() {
225                notes.push_str(rest);
226                notes.push('\n');
227            }
228            continue;
229        }
230        if in_notes {
231            notes.push_str(l);
232            notes.push('\n');
233        }
234    }
235    (purpose, notes)
236}

238/// No `purpose:` field anywhere — take the reply's last paragraph
239/// (the agent's conclusion; the preamble is reasoning) rather than
240/// the first line, which is where the monologue starts.
241fn last_paragraph(prose: &str) -> Option<String> {
242    prose
243        .trim()
244        .rsplit("\n\n")
245        .next()
246        .map(|p| {
247            p.lines()
248                .map(str::trim)
249                .filter(|l| !l.is_empty() && !l.starts_with("```"))
250                .collect::<Vec<_>>()
251                .join(" ")
252        })
253        .filter(|p| !p.is_empty())
254}

256impl Summarizer for AcpSummarizer {
257    fn kind(&self) -> &'static str {
258        "acp"
259    }
260
261    fn summarize(
262        &self,
263        module: &str,
264        path: &Path,
265        ext: &Extraction,
266        provenance: Provenance,
267    ) -> Result<ModuleCard, String> {
268        let prose = self.complete(&prompt_for(module, ext))?;
269
270        // Only the reply's closing structured block counts — a
271        // reasoning preamble must never land in `purpose`.
272        let (purpose, notes) = parse_card_body(&prose);
273        let purpose = purpose
274            .or_else(|| last_paragraph(&prose))
275            .unwrap_or_else(|| module.into());
276        Ok(ModuleCard {
277            name: module.into(),
278            path: path.to_path_buf(),
279            purpose,
280            api: ext.symbols.clone(),
281            api_omitted: 0,
282            deps: ext.edges.iter().map(|e| e.to.clone()).collect(),
283            warnings: Vec::new(),
284            notes,
285            verify: ext.files.clone(),
286            provenance,
287        })
288    }
289}
